\begin{recipe}
[
    %preparationtime = {\unit[20]{Min.}},
    bakingtime = {\unit[30]{Min.}},
    %portion = {\unit[4]{Personen}},
    calory = {ca. 8333 kcal},
    %source = {Omas Geheimrezept}
]
{Roggenschrotbrot} % Rezeptname

    % Optionale Einleitung oder Beschreibung zum Rezept
    Nach einem Rezept von Lutz Gei"sler. Angepasst an 3 Backformen 30cm von Zenker.

    % Zutatenliste
        \ingredients{
        &
	\begin{tabular}[t]{@{}ll@{}}
	\multicolumn{2}{l}{\textbf{Br"uhst"uck:}} \\
            240.65\,g & Schrot (Stufe 7.5) \\
            721.95\,ml & Wasser (kochend) \\
            48.13\,g & Salz
        \end{tabular} \\
        \addlinespace
        &
	\begin{tabular}[t]{@{}ll@{}}
	    \multicolumn{2}{l}{\textbf{Hauptteig:}} \\
            1347.63\,g & Schrot (Stufe 7.5) \\
            828.23\,g & Schrot (Stufe 3) \\
            1754.73\,ml & Wasser \\
            84.23\,g & Sauerteigstarter \\
        \end{tabular} \\
    }



    % Zubereitungsschritte
    \preparation{
        \step Zutaten f"ur das Br"uhst"uck zusammen vermischen.
        \step Folie direkt auf die Oberfl"ache der Masse legen und 8-10 Stunden bei Raumtemperatur stehen lassen.
        \step Sauerteigstarter aus dem K"uhlschrank nehmen und 1:1:1 f"uttern und (3-4 Stunden) warten bis er sich verdoppelt hat.
        \step Sauerteigstarter nochmal 1:1:1 f"uttern und (3-4 Stunden) warten bis er sich verdoppelt hat.
        \step Sauerteigstarter in eine gro"se Sch"ussel mit der Wassermenge f"ur den Hauptteig vermischen.
        \step Br"uhst"uck dazugeben und vermischen.
        \step Die restlichen Zutaten dazugeben und alles vermischen bis alles homogen ist und auf die Backformen aufteilen.
        \step Die Formen abdecken und 24 Stunden bei 18-20 °C gehen lassen.
        \step Backblech auf unterste Schiene in den Ofen einschieben und auf 250 °C vorheizen.
        \step Kastenformen mit Teig auf Gitter direckt "uber Backblech einschieben und 100\,ml kochendes Wasser auf das Blech kippen und Ofen schliessen
        \step Nach 15 Minuten den Ofen zum Dampfablassen "offnen und Backblech herausnehmen.
        \step Temperatur auf 200 °C stellen.
        \step Nach weiteren 65 Minuten Stechthermometer in Brote und Brote weiter backen bis Kerntemperatur mindestens 98 °C erreicht.
        \step Brote aus Formen nehmen und 10 Minuten weiter backen.
        \step Ofen ausmachen. Klappe leicht "offnen und Brote mind. 30 Minuten im Ofen lassen.
    }
    
    % Optionale Notiz oder Tipp am Ende des Rezepts
    \hint{Schmeckt besonders gut, wenn er noch leicht warm ist und mit einer Kugel Vanilleeis serviert wird!}

\end{recipe}
