\setRecipeMeta{Roggenschrotbrot}{3}{15 MIN}{30 MIN}{./images/cheeseburgerbowl.jpg}

\begin{recipe}
  \begin{ingredients}
  \ingredientGroup{Br"uhst"uck}
    \ingredient{171.89g Schrot grob (Stufe 7.5)}
    \ingredient{515.68g kochendes Wasser}
    \ingredient{34.38g Salz}
  \ingredientGroup{Teig}
    \ingredient{962.59g Schrot grob (Stufe 7.5)}
    \ingredient{591.59g Schrot fein (Stufe 3)}
    \ingredient{1253.38g Wasser}
    \ingredient{60.16g Roggensauerteig (aktiv)}
  \end{ingredients}
  \begin{steps}
    \step{Zutaten f"ur Br"uhst"uck in einer Schu"ssel zusammen mischen}
    \step{Sch"ussel mit Frischhaltefolie abdecken. So dass die Folie direkt auf die Oberflaeche der Masse ist. Damit sich keine Haut bildet.}
    \step{12 Stunden bei Zimmertemperatur ruhen lassen.}
    \step{Br"uhst"uck und Zutaten fuer den Teig zusammenmischen bis keine Mehlreste mehr vorhanden sind.}
    \step{3 Backformen (ca. 30cm x 26cm x 7cm) einfetten oder mit Backpapier auslegen.}
    \step{Teig auf die Backformen aufteilen.}
    \step{Formen abdecken (Duschhauben) und 24 Stunden bei 18-20 °C ruhen lassen.}
    \step{Ofen mit Ober-/Unterhitze auf 250 °C vorheizen.}
  \end{steps}
\end{recipe}
